

CREATE OR REPLACE TRIGGER DML_ADBS
BEFORE DELETE OR INSERT OR UPDATE
ON ADBS
REFERENCING NEW AS NEW OLD AS OLD

    then 
        raise_application_error('909090','Error'); 
    end if; 
end;

CREATE OR REPLACE TRIGGER LOG_CREATE_CREDIT_NOTE
AFTER INSERT
ON SALES_RETURN_MASTER REFERENCING NEW AS NEW OLD AS OLD
FOR EACH ROW
begin
insert into  log_table  VALUES (log_seq.NEXTVAL,:new.NOTE_DATE,TO_CHAR(sysdate, 'YYYY-MM-DD HH24:MI:SS'),:new.USER_ID,'Cancel','Invoice',:new.PK_ID);
end;



CREATE OR REPLACE TRIGGER LOG_CREATE_INVOICE
AFTER INSERT
ON SALES_INVOICE_MASTER REFERENCING NEW AS NEW OLD AS OLD
FOR EACH ROW
BEGIN
INSERT INTO  LOG_TABLE  VALUES (LOG_SEQ.NEXTVAL,:NEW.INVOICE_DATE,TO_CHAR(SYSDATE, 'YYYY-MM-DD HH24:MI:SS'),:NEW.USER_ID,'INSERT','INVOICE',:NEW.PK_ID);
IF :NEW.COPY_PRINTNO  <> :OLD.COPY_PRINTNO 
  THEN
INSERT INTO  LOG_TABLE  VALUES (LOG_SEQ.NEXTVAL,:NEW.INVOICE_DATE,TO_CHAR(SYSDATE, 'YYYY-MM-DD HH24:MI:SS'),:NEW.USER_ID,'PRINT','INVOICE',:NEW.PK_ID);
        END IF;
END;

CREATE OR REPLACE TRIGGER LOG_PRINT_INVOICE
AFTER INSERT OR UPDATE
ON SALES_INVOICE_MASTER 
REFERENCING NEW AS NEW OLD AS OLD
FOR EACH ROW
begin
IF :NEW.IS_PRINTED <> :OLD.IS_PRINTED
THEN
    insert into  log_table  VALUES (log_seq.NEXTVAL,to_date(to_char(sysdate,'dd/mm/yyyy'),'dd/mm/yyyy'),TO_CHAR(sysdate, 'YYYY-MM-DD HH24:MI:SS'),:new.PRINT_BY,'PRINT','INVOICE',:new.PK_ID);
END IF;
IF :NEW.COPY_PRINTNO <> :OLD.COPY_PRINTNO
  THEN
insert into  log_table  VALUES (log_seq.NEXTVAL,to_date(to_char(sysdate,'dd/mm/yyyy'),'dd/mm/yyyy'),TO_CHAR(sysdate, 'YYYY-MM-DD HH24:MI:SS'),:new.PRINT_BY,'COPY PRINT','INVOICE',:new.PK_ID);
        END IF;
end;
